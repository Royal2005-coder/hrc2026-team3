Step 4. Viết hàm pixel-to-3D 
Từ centroid pixel và depth, tính điểm 3D trong camera frame. Đây là công thức quan trọng nhất của phần RGB-D. 
import numpy as np 
 
def pixel_to_camera_point(u, v, depth_z, fx, fy, cx, cy): 
    """ 
    Convert pixel + depth to a 3D point in camera frame. 
    u, v: pixel coordinate 
    depth_z: depth value in meters 
    fx, fy, cx, cy: camera intrinsics 
    """ 
    if depth_z <= 0 or not np.isfinite(depth_z): 
        raise ValueError(f"Invalid depth value: {depth_z}") 
 
    x = (u - cx) * depth_z / fx 
    y = (v - cy) * depth_z / fy 
    z = depth_z 
    return np.array([x, y, z], dtype=float) 
 
def test_pixel_center(): 
    p = pixel_to_camera_point( 
        u=320, v=240, depth_z=1.0, 
        fx=600, fy=600, cx=320, cy=240 
    ) 
    assert np.allclose(p, [0.0, 0.0, 1.0]) 
Step 5. Lấy depth ổn định quanh centroid 
Không nên lấy depth đúng 1 pixel vì dễ nhiễu hoặc rơi vào biên object. Dùng median trong patch nhỏ hoặc trong 
mask. 
def robust_depth_from_patch(depth, u, v, radius=3): 
    h, w = depth.shape 
    u = int(round(u)) 
    v = int(round(v)) 
 
    x1 = max(0, u - radius) 
    x2 = min(w, u + radius + 1) 
    y1 = max(0, v - radius) 
    y2 = min(h, v + radius + 1) 
 
    patch = depth[y1:y2, x1:x2] 
    valid = patch[np.isfinite(patch) & (patch > 0)] 
    if len(valid) == 0: 
        return None 
    return float(np.median(valid)) 
 
def robust_depth_from_mask(depth, mask): 
    valid = depth[(mask > 0) & np.isfinite(depth) & (depth > 0)] 
    if len(valid) == 0: 
        return None 
    return float(np.median(valid))

import cv2 as cv 
import numpy as np 
 
def detect_by_color_bgr(rgb_bgr, lower_hsv, upper_hsv): 
    hsv = cv.cvtColor(rgb_bgr, cv.COLOR_BGR2HSV) 
    mask = cv.inRange(hsv, np.array(lower_hsv), np.array(upper_hsv)) 
 
    kernel = np.ones((5, 5), np.uint8) 
    mask = cv.morphologyEx(mask, cv.MORPH_OPEN, kernel) 
    mask = cv.morphologyEx(mask, cv.MORPH_CLOSE, kernel) 
 
    contours, _ = cv.findContours(mask, cv.RETR_EXTERNAL, cv.CHAIN_APPROX_SIMPLE) 
    detections = [] 
    for cnt in contours: 
        area = cv.contourArea(cnt) 
        if area < 100: 
            continue 
        x, y, w, h = cv.boundingRect(cnt) 
        m = cv.moments(cnt) 
        if m["m00"] == 0: 
            continue 
        cx = m["m10"] / m["m00"] 
        cy = m["m01"] / m["m00"] 
        detections.append({ 
            "bbox_xyxy": [x, y, x + w, y + h], 
            "centroid_px": [cx, cy], 
            "area_px": area, 
            "contour": cnt 
        }) 
    return detections, mask

def estimate_yaw_from_contour(contour): 
    rect = cv.minAreaRect(contour) 
    angle_deg = rect[-1] 
    # OpenCV angle convention can be tricky. 
    # Treat this as a grasp hint and validate visually. 
    yaw_rad = np.deg2rad(angle_deg) 
    return float(yaw_rad)

class CameraIntrinsics: 
    def __init__(self, fx, fy, cx, cy, width=None, height=None):

        self.fy = float(fy)

def robust_depth_from_patch(depth, u, v, radius=3): 
    h, w = depth.shape 
    u = int(round(u)) 
    v = int(round(v)) 
    x1, x2 = max(0, u-radius), min(w, u+radius+1) 
    y1, y2 = max(0, v-radius), min(h, v+radius+1) 
    patch = depth[y1:y2, x1:x2] 
    valid = patch[np.isfinite(patch) & (patch > 0)] 
    return None if len(valid) == 0 else float(np.median(valid))

import cv2 as cv 
import numpy as np 
from camera_utils import pixel_to_camera_point, robust_depth_from_patch 
 
def detect_by_color_bgr(rgb_bgr, lower_hsv, upper_hsv, min_area=100): 
    hsv = cv.cvtColor(rgb_bgr, cv.COLOR_BGR2HSV) 
    mask = cv.inRange(hsv, np.array(lower_hsv), np.array(upper_hsv)) 
    kernel = np.ones((5, 5), np.uint8) 
    mask = cv.morphologyEx(mask, cv.MORPH_OPEN, kernel) 
    mask = cv.morphologyEx(mask, cv.MORPH_CLOSE, kernel) 
    contours, _ = cv.findContours(mask, cv.RETR_EXTERNAL, cv.CHAIN_APPROX_SIMPLE) 
    detections = [] 
    for cnt in contours: 
        area = cv.contourArea(cnt) 
        if area < min_area: 
            continue 
        x, y, w, h = cv.boundingRect(cnt) 
        m = cv.moments(cnt) 
        if m["m00"] == 0: 
            continue 
        cx = m["m10"] / m["m00"] 
        cy = m["m01"] / m["m00"] 
        detections.append({ 
            "bbox_xyxy": [int(x), int(y), int(x+w), int(y+h)], 
            "centroid_px": [float(cx), float(cy)], 
            "area_px": float(area), 
            "contour": cnt, 
        }) 
    return detections, mask 
 
def estimate_yaw_from_contour(contour): 
    rect = cv.minAreaRect(contour) 
    return float(np.deg2rad(rect[-1]))

# Constant-velocity predictor in image space 
class SimpleTrack: 
    def __init__(self, track_id): 
        self.track_id = track_id 
        self.history = []  # list of (frame_id, x, y) 
 
    def update(self, frame_id, x, y): 
        self.history.append((frame_id, x, y)) 
        self.history = self.history[-10:] 
 
    def predict(self, target_frame): 
        if len(self.history) < 2: 
            return None 
        f0, x0, y0 = self.history[0] 
        f1, x1, y1 = self.history[-1] 
        dt = max(f1 - f0, 1) 
        vx, vy = (x1 - x0) / dt, (y1 - y0) / dt 
        return x1 + vx * (target_frame - f1), y1 + vy * (target_frame - f1)

4.1. Ảnh số, pixel và hệ tọa độ ảnh 
Ảnh RGB là ma trận H x W x 3. Mỗi pixel có tọa độ ảnh (u, v), trong đó u thường là chỉ số cột theo chiều 
ngang, v là chỉ số hàng theo chiều dọc. Gốc tọa độ ảnh thường nằm ở góc trên-trái. Với robotics, cần hết 
sức cẩn thận vì hệ tọa độ ảnh, camera frame, world frame và robot base frame không giống nhau. 
Ký hiệu Ý nghĩa Ghi chú 
u tọa độ cột pixel tăng từ trái sang phải 
v tọa độ hàng pixel tăng từ trên xuống dưới 
z hoặc depth khoảng cách đo bởi depth 
camera

bbox bounding box thường là x_min, y_min, x_max, y_max hoặc x, y, w, 
h 
mask vùng pixel thuộc vật thể cần cho centroid/pose/point cloud của vật 
 
4.2. Pinhole camera model 
Pinhole model mô tả cách điểm 3D trong camera frame chiếu lên ảnh 2D. Nếu điểm 3D trong camera 
frame là P_cam = (X, Y, Z), pixel tương ứng được tính gần đúng bởi: 
u = fx * X / Z + cx 
v = fy * Y / Z + cy 
Trong đó fx, fy là tiêu cự theo đơn vị pixel; cx, cy là tọa độ principal point/optical center. Ma trận intrinsic K 
có dạng: 
K = [[fx,  0, cx], 
     [ 0, fy, cy], 
     [ 0,  0,  1]] 
Công thức ngược, dùng nhiều nhất trong module này, là chuyển từ pixel + depth sang điểm 3D trong 
camera frame: 
X = (u - cx) * Z / fx 
Y = (v - cy) * Z / fy 
Z = depth 
4.3. Depth image và các dạng depth thường gặp 
Depth image là ảnh một kênh, mỗi pixel chứa giá trị khoảng cách. Tuy nhiên cần kiểm tra kỹ định nghĩa 
của depth vì các hệ thống có thể trả về distance-to-image-plane, distance-to-camera-center, disparity hoặc 
depth đã scale. 
Loại dữ liệu Ý nghĩa Rủi ro thường gặp 
Depth in meters Mỗi pixel là khoảng cách tính bằng mét Dễ nhầm với millimeters 
Depth in millimeters Mỗi pixel là khoảng cách tính bằng mm Nếu không chia 1000, point cloud 
phóng đại 1000 lần 
Distance to image plane Khoảng cách theo trục Z camera Phù hợp công thức pinhole đơn giản 
Euclidean distance Khoảng cách từ camera center đến 
điểm

không aligned, centroid/mask từ RGB sẽ lấy nhầm depth.

4.5. Distortion và calibration 
Camera thật thường có lens distortion, chủ yếu radial distortion và tangential distortion. Calibration tìm 
intrinsic K, distortion coefficients và đôi khi extrinsic giữa camera với một frame tham chiếu. Trong Isaac 
Sim, camera lý tưởng hơn camera thật, nhưng vẫn cần hiểu calibration để chuẩn bị cho vòng robot thật và 
để chuyển thông số OpenCV sang Isaac/Omniverse khi cần. 
Khái niệm Ý nghĩa thực dụng 
Intrinsic calibration Tìm fx, fy, cx, cy và distortion coefficients của camera. 
Extrinsic calibration Tìm pose của camera so với robot/world/base. 
Undistortion Sửa ảnh để đường thẳng ngoài đời không bị cong trên ảnh. 
Reprojection error Sai số giữa điểm 2D quan sát và điểm 2D chiếu từ mô hình camera; dùng 
để đánh giá calibration. 
 
4.6. Camera extrinsic và transform chain 
Intrinsic giúp chuyển pixel + depth sang point trong camera frame. Extrinsic giúp chuyển point đó sang 
world frame hoặc robot base frame. Theo quy ước từ Module 3, T_A_B là transform đưa tọa độ của một 
point từ frame B sang frame A. 
p_cam_h = [X, Y, Z, 1]^T 
p_world_h = T_world_cam @ p_cam_h 
p_base_h  = T_base_world @ T_world_cam @ p_cam_h 
 
# Nếu đã có T_base_cam: 
p_base_h = T_base_cam @ p_cam_h 
Đây là đường dẫn nhận thức tối thiểu cho robot: 
RGB/depth pixel -> 3D point in camera frame -> 3D point in world/base frame -> grasp/insert/place 
target

import numpy as np 
 
def pixel_to_camera(u, v, z, fx, fy, cx, cy): 
    if not np.isfinite(z) or z <= 0: 
        raise ValueError('Depth must be positive and finite') 
    X = (u - cx) * z / fx 
    Y = (v - cy) * z / fy 
    Z = z 
    return np.array([X, Y, Z], dtype=float) 
 
def camera_to_pixel(P, fx, fy, cx, cy): 
    X, Y, Z = P 
    if not np.isfinite(Z) or Z <= 0: 
        raise ValueError('Z must be positive and finite') 
    u = fx * X / Z + cx 
    v = fy * Y / Z + cy 
    return np.array([u, v], dtype=float) 
 
def make_K(fx, fy, cx, cy): 
    return np.array([[fx, 0.0, cx], [0.0, fy, cy], [0.0, 0.0, 1.0]], dtype=float)

import numpy as np 
 
def check_transform(T, atol=1e-6): 
    T = np.asarray(T, dtype=float) 
    assert T.shape == (4, 4) 
    R = T[:3, :3] 
    det_R = np.linalg.det(R) 
    ortho_err = np.linalg.norm(R.T @ R - np.eye(3)) 
    return {'det_R': float(det_R), 'ortho_err': float(ortho_err), 
            'is_valid': abs(det_R - 1.0) < 1e-3 and ortho_err < 1e-3}

import numpy as np 
 
def valid_depth_mask(depth, min_depth=0.05, max_depth=5.0): 
    depth = np.asarray(depth, dtype=float) 
    return np.isfinite(depth) & (depth >= min_depth) & (depth <= max_depth) 
 
def median_depth_in_mask(depth, mask): 
    valid = valid_depth_mask(depth) & (mask.astype(bool)) 
    values = depth[valid] 
    if values.size == 0: 
        return np.nan 
    return float(np.median(values))
